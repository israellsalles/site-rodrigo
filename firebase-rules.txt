rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /aniversariantes/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null
        && request.auth.uid == "COLE_AQUI_UID_DO_ADMIN";
    }

    match /eventos/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null
        && request.auth.uid == "COLE_AQUI_UID_DO_ADMIN";
    }

    match /configuracoes/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null
        && request.auth.uid == "COLE_AQUI_UID_DO_ADMIN";
    }
  }
}
